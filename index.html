一1 加到 N 的 Fortran 程式20 分

指揮 AI 輸出一個能做 1 加到 N 的 Fortran 程式並完成編譯，確認可以成功執行計算並輸出結果。

過程：把題目交給 AI（Claude），AI 產生下列程式；先在 Online Fortran Compiler 執行（N = 33 得到 561），再於本機 Cygwin 用 gfortran sum.f90 -o sum 編譯、./sum 執行，輸入 N = 10 得到 55，與公式 N(N+1)/2 一致。總和用 integer(kind=8) 儲存，避免 N 較大時溢位；N < 1 時顯示錯誤訊息。

program sum_1_to_n
  implicit none
  integer :: n, i
  integer(kind=8) :: total

  print *, 'Enter N:'
  read *, n
  if (n < 1) then
    print *, 'N must be a positive integer.'
    stop
  end if

  total = 0
  do i = 1, n
    total = total + i
  end do

  print *, 'Sum of 1 to', n, '=', total
end program sum_1_to_n

! 計算 1 + 2 + ... + N 的總和
program sum_1_to_n
  implicit none                 ! 所有變數必須明確宣告
  integer :: n, i               ! n：上限；i：迴圈計數器
  integer(kind=8) :: total      ! 累加結果，8 位元組避免溢位

  print *, 'Enter N:'           ! 提示輸入 N
  read *, n                     ! 讀入 N
  if (n < 1) then               ! 檢查是否為正整數
    print *, 'N must be a positive integer.'  ! 錯誤訊息
    stop                        ! 結束程式
  end if                        ! 結束判斷

  total = 0                     ! 累加器歸零
  do i = 1, n                   ! 從 1 迴圈到 N
    total = total + i           ! 把 i 加進總和
  end do                        ! 迴圈結束

  print *, 'Sum of 1 to', n, '=', total  ! 輸出結果
end program sum_1_to_n          ! 程式結束
